## Geändert
- portalAsync hat die funktionsprüfung auf die Spielregel 'allow_entering_nether_using_portals' bekommen.

(Wenn die gamerule auf false steht, können Nether Portale aktiviert werden, aber sie teleortieren nicht den spieler.)

```java title="tessera\folia-server\src\minecraft\java\net\minecraft\world\level\block\NetherPortalBlock.java"
    @Override
    public boolean portalAsync(ServerLevel sourceWorld, Entity portalTarget, BlockPos portalPos) {
        if (!ca.spottedleaf.moonrise.common.util.TickThread.isTickThreadFor(portalTarget)) {
            return false;
        }
        if (!ca.spottedleaf.moonrise.common.util.TickThread.isTickThreadFor(sourceWorld, portalPos)) {
            return false;
        }
        // tessera 26.2 start check for gamerule (are we allowed to travel using portals)
        //  moved after the code for checking if thats for a player
        if(!sourceWorld.getGameRules().get(GameRules.ALLOW_ENTERING_NETHER_USING_PORTALS)){
            if(portalTarget instanceof Player player) 
                punish(player, sourceWorld); // implementation not included
            return false;
        }
        // tessera 26.2 end check for gamerule (are we allowed to travel using portals)
        return portalTarget.netherPortalLogicAsync(portalPos);
    }
```
